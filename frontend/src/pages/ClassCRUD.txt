import { useState, useEffect } from "react";

export default function ClassCRUD() {
  // Dummy data to populate the deep sea
  const [classes, setClasses] = useState([
    { id: 1, name: "Abyssal Biology 101" },
    { id: 2, name: "Advanced Hydro-Tech" },
    { id: 3, name: "Bioluminescent Design" },
  ]);
  const [name, setName] = useState("");

  const addClass = () => {
    if (!name) return;
    setClasses([...classes, { id: Date.now(), name }]);
    setName("");
  };

  const removeClass = (id) => {
    setClasses(classes.filter((c) => c.id !== id));
  };

  

  return (
    <div className="p-8 bg-[#020c1b] min-h-screen text-white font-mono">
      {/* Header with Neo-Brute underline */}
      <div className="inline-block mb-10">
        <h1 className="text-3xl font-black uppercase italic tracking-tighter text-cyan-400">
          Class/Course Management
        </h1>
        <div className="h-2 w-full bg-cyan-900 mt-1 shadow-[4px_4px_0px_0px_#0891b2]" />
      </div>

      {/* Input Section */}
      <div className="flex flex-wrap gap-4 mb-12">
        <input
          type="text"
          placeholder="ENTER CLASS IDENTIFIER..."
          value={name}
          onChange={(e) => setName(e.target.value)}
          className="bg-white/5 border-2 border-cyan-900 p-3 rounded-none focus:outline-none focus:border-cyan-400 focus:bg-white/10 transition-all text-cyan-100 placeholder:text-cyan-900 w-72 backdrop-blur-md"
        />
        <button 
          onClick={addClass} 
          className="bg-cyan-500 text-black font-black uppercase px-6 py-3 border-2 border-black shadow-[4px_4px_0px_0px_#0891b2] hover:translate-x-[2px] hover:translate-y-[2px] hover:shadow-none transition-all active:bg-white"
        >
          Add Class +
        </button>
      </div>

      {/* Class List */}
      <div className="grid gap-4 max-w-2xl">
        {classes.length > 0 ? (
          classes.map((c) => (
            <div 
              key={c.id} 
              className="group flex justify-between items-center p-4 bg-white/5 border-2 border-cyan-900/50 backdrop-blur-sm hover:border-cyan-400 transition-colors relative"
            >
              {/* Subtle accent bar for active-style feel */}
              <div className="absolute left-0 top-0 bottom-0 w-1 bg-cyan-900 group-hover:bg-cyan-400" />
              
              <span className="pl-4 font-bold text-lg tracking-wide uppercase group-hover:text-cyan-200">
                {c.name}
              </span>

              <button 
                onClick={() => removeClass(c.id)} 
                className="text-[10px] font-black uppercase tracking-tighter border border-red-900 px-2 py-1 text-red-500 hover:bg-red-500 hover:text-black transition-colors"
              >
                Terminate [X]
              </button>
            </div>
          ))
        ) : (
          <div className="border-2 border-dashed border-cyan-900 p-8 text-center text-cyan-900 uppercase font-black">
            No active signals detected...
          </div>
        )}
      </div>
    </div>
  );
}
